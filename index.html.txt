    <!-- SCRIPT DA APLICAÇÃO -->
    <script>
        // Mapeamento oficial de Centros e seus Depósitos
        const MAPPING_DEPOSITOS = {
            '1000': [
                { code: 'AA01', name: 'AA01 (Primavera)' },
                { code: 'AA02', name: 'AA02 (VV6)' },
                { code: 'AA11', name: 'AA11 (Pneus)' },
                { code: 'AI01', name: 'AI01 (Indústria)' }
            ],
            '2000': [
                { code: 'AA01', name: 'AA01 (Agrícola)' },
                { code: 'AA03', name: 'AA03 (Pátio)' },
                { code: 'AA11', name: 'AA11 (Pneus)' },
                { code: 'AI01', name: 'AI01 (Indústria)' }
            ],
            '3000': [
                { code: 'AA01', name: 'AA01 (Agrícola)' },
                { code: 'AA11', name: 'AA11 (Pneus)' },
                { code: 'AI01', name: 'AI01 (Indústria)' }
            ],
            '4000': [
                { code: 'AA01', name: 'AA01 (Agrícola)' },
                { code: 'AA11', name: 'AA11 (Pneus)' },
                { code: 'AI01', name: 'AI01 (Indústria)' }
            ]
        };

        let materialsData = [];

        function handleFileUpload(e) {
            const file = e.target.files[0];
            if (!file) return;

            const reader = new FileReader();
            reader.onload = function(evt) {
                const data = new Uint8Array(evt.target.result);
                const workbook = XLSX.read(data, {type: 'array'});
                const firstSheetName = workbook.SheetNames[0];
                const worksheet = workbook.Sheets[firstSheetName];
                const json = XLSX.utils.sheet_to_json(worksheet, {header: 1});
                
                processJsonData(json);
            };
            reader.readAsArrayBuffer(file);
        }

        function processPastedData() {
            const raw = document.getElementById('pasteArea').value.trim();
            if (!raw) return;

            const lines = raw.split('\n');
            const json = lines.map(line => line.split('\t').length > 1 ? line.split('\t') : line.split(';'));
            processJsonData(json);
        }

        function processJsonData(rows) {
            if (rows.length === 0) return;

            materialsData = [];
            // Assumindo linha de cabeçalho
            let startIdx = 1;
            
            for (let i = startIdx; i < rows.length; i++) {
                const row = rows[i];
                if (!row || row.length === 0) continue;

                const material = row[0] || row[1] || '';
                const descricao = row[1] || row[2] || 'Material sem descrição';
                const centro = String(row[2] || row[0] || '1000').trim();
                const deposito = String(row[3] || 'AA01').trim();
                const segmento = row[4] || 'AGRÍCOLA GENERAL';

                if (material) {
                    materialsData.push({
                        id: i,
                        material: String(material).trim(),
                        descricao: String(descricao).trim(),
                        centro: ['1000','2000','3000','4000'].includes(centro) ? centro : '1000',
                        deposito: deposito || 'AA01',
                        segmento: String(segmento).trim()
                    });
                }
            }

            updateCounters();
            filterData();
        }

        function updateCounters() {
            const counts = { '1000': 0, '2000': 0, '3000': 0, '4000': 0 };
            materialsData.forEach(item => {
                if (counts[item.centro] !== undefined) counts[item.centro]++;
            });

            document.getElementById('count-1000').innerText = `${counts['1000']} itens`;
            document.getElementById('count-2000').innerText = `${counts['2000']} itens`;
            document.getElementById('count-3000').innerText = `${counts['3000']} itens`;
            document.getElementById('count-4000').innerText = `${counts['4000']} itens`;

            document.getElementById('totalCount').innerText = materialsData.length;
        }

        function filterData() {
            const search = document.getElementById('searchInput').value.toLowerCase();
            const centro = document.getElementById('filterCentro').value;
            const deposito = document.getElementById('filterDeposito').value;

            const filtered = materialsData.filter(item => {
                const matchSearch = item.material.toLowerCase().includes(search) || item.descricao.toLowerCase().includes(search);
                const matchCentro = !centro || item.centro === centro;
                const matchDeposito = !deposito || item.deposito === deposito;
                return matchSearch && matchCentro && matchDeposito;
            });

            renderTable(filtered);
            document.getElementById('displayedCount').innerText = filtered.length;
        }

        function renderTable(data) {
            const tbody = document.getElementById('tableBody');
            tbody.innerHTML = '';

            if (data.length === 0) {
                tbody.innerHTML = `<tr><td colspan="6" class="p-8 text-center text-slate-400">Nenhum registro encontrado.</td></tr>`;
                return;
            }

            data.forEach(item => {
                const tr = document.createElement('tr');
                tr.className = 'hover:bg-slate-50 transition';

                const depositosDisponiveis = MAPPING_DEPOSITOS[item.centro] || MAPPING_DEPOSITOS['1000'];
                let selectOptions = depositosDisponiveis.map(d => 
                    `<option value="${d.code}" ${item.deposito === d.code ? 'selected' : ''}>${d.name}</option>`
                ).join('');

                tr.innerHTML = `
                    <td class="p-3 font-semibold text-slate-700">${item.material}</td>
                    <td class="p-3 text-slate-600">${item.descricao}</td>
                    <td class="p-3"><span class="px-2 py-0.5 rounded text-[11px] font-bold bg-slate-200 text-slate-700">${item.centro}</span></td>
                    <td class="p-3">
                        <select onchange="updateItemDeposito(${item.id}, this.value)" class="p-1 border border-slate-300 rounded text-xs focus:ring-1 focus:ring-emerald-500">
                            ${selectOptions}
                        </select>
                    </td>
                    <td class="p-3 text-slate-500">${item.segmento}</td>
                    <td class="p-3 text-center">
                        <button onclick="removeItem(${item.id})" class="text-rose-500 hover:text-rose-700 p-1">
                            <i class="fa-solid fa-trash"></i>
                        </button>
                    </td>
                `;
                tbody.appendChild(tr);
            });
        }

        function updateItemDeposito(id, val) {
            const item = materialsData.find(m => m.id === id);
            if (item) item.deposito = val;
        }

        function removeItem(id) {
            materialsData = materialsData.filter(m => m.id !== id);
            updateCounters();
            filterData();
        }

        function exportData(format) {
            if (materialsData.length === 0) {
                alert('Carregue os dados antes de exportar.');
                return;
            }

            if (format === 'csv') {
                let csvContent = "data:text/csv;charset=utf-8,Material;Texto;Centro;Deposito;Segmento\n";
                materialsData.forEach(r => {
                    csvContent += `${r.material};"${r.descricao}";${r.centro};${r.deposito};"${r.segmento}"\n`;
                });
                const encodedUri = encodeURI(csvContent);
                const link = document.createElement("a");
                link.setAttribute("href", encodedUri);
                link.setAttribute("download", `CARGA_SAP_MATERIAIS_MPR_${new Date().toISOString().slice(0,10)}.csv`);
                document.body.appendChild(link);
                link.click();
                document.body.removeChild(link);
            } else {
                const ws = XLSX.utils.json_to_sheet(materialsData.map(m => ({
                    'Material': m.material,
                    'Descrição': m.descricao,
                    'Centro': m.centro,
                    'Depósito': m.deposito,
                    'Segmento': m.segmento
                })));
                const wb = XLSX.utils.book_new();
                XLSX.utils.book_append_sheet(wb, ws, "Materiais_MPR");
                XLSX.writeFile(wb, `CARGA_MATERIAIS_MPR_${new Date().toISOString().slice(0,10)}.xlsx`);
            }
        }
